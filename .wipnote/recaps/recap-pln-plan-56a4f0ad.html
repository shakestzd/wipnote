+ // labelHerdrPane names the caller's herdr pane "<id>: <title>". Best effort:
+ // outside herdr nothing runs, and a missing, failing or hung herdr is ignored
+ // once herdr.LabelTimeout has passed, so a launch or start never fails on it.
+ func labelHerdrPane(id, title string) {
+ 	env := herdr.EnvFrom(os.Getenv)
+ 	if !env.InHerdr || env.CallerPane == "" || id == "" {
+ 		return
+ 	}
+ 	c := herdr.New(env)
+ 	c.Timeout = herdr.LabelTimeout
+ 	_ = herdr.LabelCallerPane(context.Background(), c, env, herdr.WorkLabel(id, title))
+ }

+ // labelHerdrPaneForWorkItem labels the caller's herdr pane for a launch's
+ // work item, reading its title from projectRoot's .wipnote/. Outside herdr
+ // or with no work item it reads nothing. An item that cannot be read is
+ // labelled with its id alone.
+ func labelHerdrPaneForWorkItem(projectRoot, id string) {
+ 	if id == "" || !herdr.EnvFrom(os.Getenv).InHerdr {
+ 		return
+ 	}
+ 	title := ""
+ 	if projectRoot != "" {
+ 		id, title = workItemTitle(filepath.Join(projectRoot, ".wipnote"), id)
+ 	}
+ 	labelHerdrPane(id, title)
+ }

+ // workItemTitle resolves id in wipnoteDir and returns its full id and title.
+ // An item that cannot be read gives the id back and no title.
+ func workItemTitle(wipnoteDir, id string) (string, string) {
+ 	if id == "" {
+ 		return "", ""
+ 	}
+ 	n, err := resolveNodeByUnionID(wipnoteDir, id)
+ 	if err != nil || n == nil {
+ 		return id, ""
+ 	}
+ 	return n.ID, n.Title
+ }

+ // ReadTranscriptChunk reads at most max bytes of the transcript from offset
+ // and returns only the complete lines in that window, plus the offset to
+ // resume from. It is the bounded tail reader the Stop and UserPromptSubmit
+ // hooks use: each pass reads what was appended since the last one.
+ //
+ //   - An offset past the end (the transcript was rewritten) restarts at 0.
+ //   - An offset in the middle of a line skips to the next line start.
+ //   - A window with no line end at all is skipped whole, so a line longer
+ //     than max can never stall the reader.
+ func ReadTranscriptChunk(path string, offset, max int64) ([]byte, int64, error) {
+ 	f, err := os.Open(path)
+ 	if err != nil {
+ 		return nil, offset, err
+ 	}
+ 	defer f.Close()
+ 	info, err := f.Stat()
+ 	if err != nil {
+ 		return nil, offset, err
+ 	}
+ 	if offset < 0 || offset > info.Size() {
+ 		offset = 0
+ 	}
+ 	size := min(info.Size()-offset, max)
+ 	if size <= 0 {
+ 		return nil, offset, nil
+ 	}
+ 	buf := make([]byte, size)
+ 	if _, err := f.ReadAt(buf, offset); err != nil && err != io.EOF {
+ 		return nil, offset, err
+ 	}
+ 	start := 0
+ 	if midLine(f, offset) {
+ 		nl := bytes.IndexByte(buf, '\n')
+ 		if nl < 0 {
+ 			return nil, offset + size, nil
+ 		}
+ 		start = nl + 1
+ 	}
+ 	end := bytes.LastIndexByte(buf, '\n') + 1
+ 	if end <= start {
+ 		if size == max {
+ 			return nil, offset + size, nil
+ 		}
+ 		return nil, offset + int64(start), nil
+ 	}
+ 	return buf[start:end], offset + int64(end), nil
+ }

+ // midLine reports whether offset falls inside a line: the byte before it is
+ // not a newline.
+ func midLine(f *os.File, offset int64) bool {
+ 	if offset == 0 {
+ 		return false
+ 	}
+ 	var b [1]byte
+ 	if _, err := f.ReadAt(b[:], offset-1); err != nil {
+ 		return false
+ 	}
+ 	return b[0] != '\n'
+ }
